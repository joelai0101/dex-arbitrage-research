cmake_minimum_required(VERSION 3.16)
project(trader_paper_reconstruction LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_library(paper_core paper_batch_scheduler.cpp paper_batch_reference_model.cpp paper_batch_layerwise_model.cpp paper_edge_grouping.cpp)
target_include_directories(paper_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(trader_paper_eg stream_driver.cpp)
target_link_libraries(trader_paper_eg PRIVATE paper_core)
if(WIN32)
  target_link_libraries(trader_paper_eg PRIVATE psapi)
endif()
foreach(test IN ITEMS paper_batch_scheduler paper_batch_layerwise_model paper_batch_order_independence paper_edge_grouping)
  add_executable(test_${test} test_${test}.cpp)
  target_link_libraries(test_${test} PRIVATE paper_core)
  add_test(NAME ${test} COMMAND test_${test})
endforeach()

cmake_minimum_required(VERSION 3.16)
project(trader_paper_faithful LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(TRADER_PROFILE "Separate diagnostic phase timing" OFF)
add_library(paper_support paper_batch_scheduler.cpp paper_batch_reference_model.cpp)
target_include_directories(paper_support PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(TRADER_PROFILE)
  target_compile_definitions(paper_support PUBLIC TRADER_PROFILE)
endif()
add_executable(test_faithful test_faithful.cpp)
target_link_libraries(test_faithful PRIVATE paper_support)
add_executable(trader_faithful stream_driver.cpp)
target_link_libraries(trader_faithful PRIVATE paper_support)
if(WIN32)
  target_link_libraries(trader_faithful PRIVATE psapi)
endif()
enable_testing()
add_test(NAME faithful COMMAND test_faithful)
